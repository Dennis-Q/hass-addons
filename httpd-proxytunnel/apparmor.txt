#include <tunables/global>

profile httpd-proxytunnel flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities: Apache binds 443 as root, then drops to the apache user
  capability chown,
  capability dac_override,
  capability fowner,
  capability fsetid,
  capability kill,
  capability net_bind_service,
  capability setgid,
  capability setuid,

  signal (send) set=(kill,term,int,hup,cont,usr1,winch),
  signal (receive),

  # Proxy traffic, DNS lookups of the allowed host, HIBP password check
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,

  # S6-Overlay
  /init rix,
  /bin/** ix,
  /usr/bin/** ix,
  /usr/sbin/** ix,

  # Brute-force protection (a script: bash must be able to read it)
  /usr/bin/proxytunnel-ban rix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** rix,
  /command/** ix,
  /etc/s6-overlay/** rix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Libraries, Apache modules and configuration
  /lib/** mr,
  /usr/lib/** mr,
  /etc/** r,
  /etc/apache2/conf.d/20-proxy-vhost.conf w,
  /etc/apache2/.htpasswd w,
  /var/www/** r,

  # Options, self-signed certificate (/data) and own certificate (/ssl, read-only)
  /data/{,**} rw,
  /ssl/{,**} r,

  # Logging to the add-on log
  /proc/** r,
  @{PROC}/@{pid}/fd/* rw,
  /dev/null rw,
}
